\chapter{Global Notation Glossary}
\label{app:global_glossary}

This appendix contains a global glossary of the main symbols used throughout the thesis.
Its purpose is to keep the notation consistent across the analytical background, the supervised ANN branch, the PINN branch, and the benchmark appendices.

\begin{table}[H]
    \ttabbox[\FBwidth]
    {\caption{Global notation glossary: core pricing and model notation}}
    {\begin{tabular}{|P{4.0cm}|P{7.8cm}|}
        \hline
        \textbf{Symbol} & \textbf{Meaning} \\
        \hline
        $t,\;T,\;\tau$ & Valuation time, maturity, and time to maturity ($\tau=T-t$). \\
        \hline
        $S_t,\;K,\;m_0,\;m,\;x$ & Spot price, strike, fixed-date moneyness \(m_0=S_0/K\), PDE moneyness \(m=S/K\), and log-moneyness \(x=\log m\). \\
        \hline
        $r$ & Constant risk-free interest rate. \\
        \hline
        $\mathbb Q$ & Risk-neutral probability measure. \\
        \hline
        $V_t$ & Arbitrage-free option value at time $t$. \\
        \hline
        $V(t_0,x)$ & Option value at valuation time $t_0$ when the state variable is $x$. \\
        \hline
        $V_{BS},\;V_{\text{target}}$ & Black-Scholes option value and target option value used in IV inversion. \\
        \hline
        $\Psi,\;\Psi_{\mathrm{put}}$ & Option payoff function at maturity; for the thesis pricing target, $\Psi_{\mathrm{put}}=(K-S_T)^+$. \\
        \hline
        $\theta$ & Generic model-parameter vector. \\
        \hline
        $\kappa,\;\bar\nu,\;\gamma,\;\rho,\;\nu_0$ & Heston parameters: mean reversion, long-run variance, vol-of-vol, correlation, and initial variance. Later chapters also use $\bar v$ and $v_0$ for the variance parameters. \\
        \hline
        $\nu_t,\;v,\;\bar v,\;v_0$ & Variance process and implementation notation for the PINN/ANN chapters: current variance state, long-run variance, and initial variance. \\
        \hline
        \multicolumn{2}{l}{Source: Author's elaboration}
    \end{tabular}}
\end{table}

\begin{table}[H]
    \ttabbox[\FBwidth]
    {\caption{Global notation glossary: ANN and CaNN notation}}
    {\begin{tabular}{|P{4.0cm}|P{7.8cm}|}
        \hline
        \textbf{Symbol} & \textbf{Meaning} \\
        \hline
        ANN baseline, ANN Sobolev & Supervised implied-volatility surrogates used in the ANN branch. Both use the same input-output map; the Sobolev variant augments the baseline value loss with IV-derivative information. \\
        \hline
        CaNN baseline, CaNN Sobolev & Calibration workflows that use a frozen ANN surrogate as fast forward evaluator in the inverse Heston calibration problem. The two variants keep the same calibration objective and differ only in whether the evaluator is the ANN baseline or the ANN Sobolev surrogate. \\
        \hline
        $f_{NN},\;\mathbf{x},\;\mathbf{w},\;\hat\sigma_{\mathrm{imp}}$ & ANN implied-volatility map, input vector, trainable weights and biases, and predicted implied volatility. In Chapter~\ref{ch:ann_surrogate}, \(\mathbf{x}=(\rho,\kappa,\gamma,\bar v,v_0,m,\tau,r)\). \\
        \hline
        $\hat V_{\mathrm{ANN}},\;\mathrm{Greek}_{\mathrm{ANN}}(q)$ & Price and sensitivity obtained by substituting the ANN predicted implied volatility into the Black-Scholes formula and differentiating the resulting price with respect to input direction \(q\). \\
        \hline
        $\mathcal{L}_{\mathrm{MSE}}$ & Baseline ANN mean-squared-error loss in implied-volatility space. \\
        \hline
        $\mathcal{L}_{\mathrm{Sob}},\;\mathcal{L}_{\mathrm{ANN\text{-}Sob}}$ & Sobolev derivative loss for IV-derivative targets and total Sobolev ANN objective. \\
        \hline
        $\mathcal{J},\;d_{i,q},\;s_q$ & Sobolev derivative directions, reference IV derivative target at anchor point \(i\) in direction \(q\), and robust scale used to normalize that derivative direction. \\
        \hline
        $B,\;B_{\mathrm{Sob}},\;\lambda_{\mathrm{Sob}}$ & Baseline mini-batch size, Sobolev mini-batch size, and weight of the Sobolev derivative loss in the ANN Sobolev objective. \\
        \hline
        \multicolumn{2}{l}{Source: Author's elaboration}
    \end{tabular}}
\end{table}

\begin{table}[H]
    \ttabbox[\FBwidth]
    {\caption{Global notation glossary: PINN notation}}
    {\begin{tabular}{|P{4.0cm}|P{7.8cm}|}
        \hline
        \textbf{Symbol} & \textbf{Meaning} \\
        \hline
        PINN baseline, PINN Sobolev, PINN Sobolev+ACV & Physics-informed price-surface variants. The baseline PINN is the PDE-trained pricing model; the Sobolev variant adds derivative targets; the ACV variant adds an analytic control-variate correction for the hard short-maturity ATM region. \\
        \hline
        $u_\phi,\;V_\phi$ & Neural price surface produced by the PINN, with trainable parameters $\phi$. \\
        \hline
        $\mathcal{R}_\phi$ & Heston PDE residual evaluated on the PINN output. \\
        \hline
        $\mathcal{S}_D,\;\mathcal{S}_T,\;\mathcal{S}_{\mathrm{low}}$ & Interior, terminal, and lower-boundary collocation sets used in baseline PINN training. \\
        \hline
        $\mathcal{L}_{\mathrm{PDE}},\;\mathcal{L}_{\mathrm{term}},\;\mathcal{L}_{\mathrm{bc}}$ & Baseline PINN loss terms for the interior PDE residual, terminal payoff, and boundary condition. \\
        \hline
        $\mathcal{L}_{\mathrm{PINN}},\;\mathcal{L}_{\mathrm{base}},\;\mathcal{L}_{\mathrm{Sob}}$ & Total baseline PINN objective, weighted baseline objective used in Sobolev refinement, and total Sobolev PINN objective. \\
        \hline
        $\mathcal{G}^{\star}(x),\;\mathcal{G}_{\phi}(x)$ & Reference Greek vector and PINN autodiff Greek vector at input point $x$. \\
        \hline
        $\mathcal{S}_{\mathrm{g}},\;\mathcal{Q}_1$ & Greek anchor set and set of first-order Greek directions used in the Sobolev PINN loss. \\
        \hline
        $q_{\phi}(x),\;q^{\star}(x)$ & PINN autodiff sensitivity and semi-analytical Heston reference sensitivity for Greek direction \(q\). \\
        \hline
        $\mathcal{L}_{\mathrm{g},1},\;\mathcal{L}_{\mathrm{g},2}$ & Sobolev Greek loss terms for first-order sensitivities and Gamma supervision. \\
        \hline
        $s_q,\;s_{\mathrm{Gamma}},\;\varepsilon$ & Scale factors used to normalize Sobolev Greek errors and numerical stabilizer used in the denominator. \\
        \hline
        $\begin{gathered}\lambda_{\mathrm{PDE}},\;\lambda_{\mathrm{term}},\;\lambda_{\mathrm{bc}}\\ \lambda_{\mathrm{g},1},\;\lambda_{\mathrm{g},2}\end{gathered}$ & Loss weights for the PINN baseline terms and the Sobolev derivative terms. \\
        \hline
        $V_{\mathrm{PINN}},\;V_{\mathrm{ref}}$ & PINN price and numerical reference price used in pricing benchmark residuals. \\
        \hline
        $p_{\alpha}$ & Empirical \(\alpha\)-quantile of a reported error distribution; for example, \(p_{95}\) and \(p_{99}\). \\
        \hline
        \multicolumn{2}{l}{Source: Author's elaboration}
    \end{tabular}}
\end{table}

\begin{table}[H]
    \ttabbox[\FBwidth]
    {\caption{Global notation glossary: analytical control variate notation}}
    {\begin{tabular}{|P{4.0cm}|P{7.8cm}|}
        \hline
        \textbf{Symbol} & \textbf{Meaning} \\
        \hline
        $z$ & PINN input vector, usually \(z=(\tau,m,v,\rho,\kappa,\gamma,\bar v,r)\). \\
        \hline
        $V_{\mathrm{Sob}}(z)$ & Price produced by the Sobolev PINN before the analytical control-variate correction. \\
        \hline
        $V_{\mathrm{ACV}}(z)$ & Corrected price surface obtained by blending the Sobolev PINN with a local analytical component. \\
        \hline
        $V_{\mathrm{BS}}^{\mathrm{put}}(\tau,x,v,r)$ & Local Black-Scholes put component used inside the ACV correction, with \(x=\log m\) and local variance level \(v\). \\
        \hline
        $\chi(\tau,x)$ & Smooth localization gate that activates the analytical correction near the short-maturity at-the-money region. \\
        \hline
        $\sigma(\cdot)$ & Logistic function used to build the ACV localization gate. \\
        \hline
        $x_c,\;\tau_c$ & Half-width parameters controlling the active log-moneyness and maturity region of the ACV gate. \\
        \hline
        $\delta_x,\;\delta_\tau$ & Smoothness parameters controlling the transition of the ACV gate in log-moneyness and maturity. \\
        \hline
        $\varepsilon_{\mathrm{BS}}$ & Numerical floor used in the local Black-Scholes component to avoid division by zero when \(v\tau\) is very small. \\
        \hline
        $\mathcal{D}_{\mathrm{hard}}$ & Short-maturity at-the-money diagnostic region used to evaluate the ACV correction. \\
        \hline
        \multicolumn{2}{l}{Source: Author's elaboration}
    \end{tabular}}
\end{table}

        \hline
        $\sigma_{imp}$ & Implied volatility obtained from numerical inversion. \\

        \hline
        $\hat z_i,\;z_i^{\mathrm{ref}}$ & Model and reference values for a generic benchmark target at evaluation point $i$. \\
        \hline
        $\mathrm{MSE},\;\mathrm{RMSE},\;\mathrm{MAPE}$ & Mean squared error, root mean squared error, and mean absolute percentage error used for benchmark reporting. \\
        \hline
        $\Pi_j,\;\Psi_j$ & Heston call-pricing probability terms and their Fourier integrands. \\
        \hline
        $\begin{gathered}\mathrm{Delta},\;\mathrm{Gamma},\;\mathrm{Vega}\\ \mathrm{Rho},\;\mathrm{Theta}\end{gathered}$ & Standard option Greeks: spot first derivative, spot second derivative, variance/volatility sensitivity, interest-rate sensitivity, and time sensitivity. \\
