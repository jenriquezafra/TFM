\section{Methodology}

\begin{frame}{Supervised branch: ANN/CaNN}
  \centering
  \workflowdiagram{1}{0.23}

  \vspace{4mm}
  \begin{itemize}
    \item The ANN learns the Heston implied-volatility surface from COS/Brent labels.
    \item Sobolev augmentation adds derivative information to improve the local geometry of the learned IV surface.
    \item CaNN uses the frozen ANN as a fast evaluator inside the inverse calibration problem.
  \end{itemize}
\end{frame}

\begin{frame}{Physics-informed branch: PINN}
  \centering
  \workflowdiagram{0.23}{1}

  \vspace{4mm}
  \begin{itemize}
    \item The baseline PINN learns the put price surface from the Heston PDE, terminal payoff, and boundary conditions.
    \item Sobolev PINN adds Greek targets and is interpreted through a pricing--Greek trade-off.
    \item Sobolev+ACV is a local analytical correction for the short-maturity ATM region.
  \end{itemize}
\end{frame}
