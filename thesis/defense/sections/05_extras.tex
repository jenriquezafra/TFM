\begin{frame}{Heston dynamics}
  \[
  \begin{aligned}
    dS_t &= rS_t\,dt+\sqrt{v_t}S_t\,dW_t^{S},\\
    dv_t &= \kappa(\bar v-v_t)\,dt+\gamma\sqrt{v_t}\,dW_t^{v},\\
    d\langle W^{S},W^{v}\rangle_t &= \rho\,dt.
  \end{aligned}
  \]

  \begin{itemize}
    \item Parameters: \(\rho\) correlation, \(\kappa\) mean reversion, \(\gamma\) vol-of-vol, \(\bar v\) long-run variance, \(v_0\) initial variance.
    \item The stochastic variance component generates non-flat implied-volatility surfaces.
  \end{itemize}
\end{frame}

\begin{frame}{Heston pricing PDE}
  \[
  \frac{\partial V}{\partial t}
  + rS\frac{\partial V}{\partial S}
  + \kappa(\bar v-v)\frac{\partial V}{\partial v}
  + \frac{1}{2}vS^2\frac{\partial^2 V}{\partial S^2}
  + \rho\gamma vS\frac{\partial^2 V}{\partial S\partial v}
  + \frac{1}{2}\gamma^2v\frac{\partial^2 V}{\partial v^2}
  - rV = 0.
  \]

  \begin{itemize}
    \item The PINN branch enforces this equation in the transformed pricing variables.
    \item Terminal payoff and boundary terms complete the baseline loss.
  \end{itemize}
\end{frame}

\begin{frame}{COS and IV inversion}
  \begin{itemize}
    \item COS pricing uses the Heston characteristic function on a finite truncation interval.
    \item Main reference configuration: \(N=1500\), \(L=50\), fallback \(L=80\) when stability checks require it.
    \item Implied volatility is recovered with Brent inversion, tolerance \(10^{-8}\), and bounded search.
  \end{itemize}
\end{frame}

\begin{frame}{ANN Sobolev objective}
  \[
    \mathcal{L}_{\mathrm{ANN}}
      =
      \frac{1}{n}\sum_i
      \left|\hat\sigma_{\mathrm{IV}}(x_i)-\sigma_{\mathrm{IV}}^{\mathrm{ref}}(x_i)\right|^2
      +
      \lambda_{\mathrm{S}}
      \frac{1}{n}\sum_i
      \left\|\nabla_x\hat\sigma_{\mathrm{IV}}(x_i)
      - \nabla_x\sigma_{\mathrm{IV}}^{\mathrm{ref}}(x_i)\right\|^2.
  \]

  \begin{itemize}
    \item The baseline learns pointwise IV labels.
    \item The Sobolev variant also constrains the local derivative structure of the IV surface.
  \end{itemize}
\end{frame}

\begin{frame}{CaNN inverse problem}
  \[
    \theta^\star
    =
    \arg\min_{\theta\in\Theta}
    \frac{1}{N_q}\sum_{j=1}^{N_q}
    \left(
      \hat\sigma_{\mathrm{IV}}(m_j,\tau_j,r_j;\theta)
      -
      \sigma^{\mathrm{mkt}}_{\mathrm{IV},j}
    \right)^2.
  \]

  \begin{itemize}
    \item \(N_q=35\) quotes on a \(5\times7\) moneyness--maturity grid.
    \item The forward evaluator is the frozen ANN baseline or ANN Sobolev surrogate.
    \item Search uses Differential Evolution with local polishing.
  \end{itemize}
\end{frame}

\begin{frame}{PINN loss decomposition}
  \[
    \mathcal{L}_{\mathrm{PINN}}
    =
    \lambda_D\mathcal{L}_{D}
    +
    \lambda_T\mathcal{L}_{T}
    +
    \lambda_B\mathcal{L}_{B}.
  \]

  \begin{itemize}
    \item \(\mathcal{L}_D\): Heston PDE residual on interior collocation points.
    \item \(\mathcal{L}_T\): terminal payoff condition.
    \item \(\mathcal{L}_B\): boundary constraints.
    \item Sobolev PINN adds supervised derivative terms for Greek accuracy.
  \end{itemize}
\end{frame}

\begin{frame}{Analytical control variate}
  \[
    V_{\mathrm{ACV}}(x)
    =
    g(x)V_{\mathrm{BS}}(x)
    +
    \bigl(1-g(x)\bigr)V_{\mathrm{PINN}}(x).
  \]

  \begin{itemize}
    \item \(g(x)\) is a smooth gate concentrated near short maturities and \(m\approx1\).
    \item The diagnostic isolates the contribution of the analytical component in the hard region.
  \end{itemize}
\end{frame}
